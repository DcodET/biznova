# ==============================================================================
# BIZNOVA Platform - Enterprise Security & Protection Configuration
# Apache & LiteSpeed Web Server Hardening Rules
# ==============================================================================

# 1. Force HTTPS Encryption
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]
</IfModule>

# 2. Disable Directory Browsing
Options -Indexes

# 3. Block Direct Access to Sensitive System & Environment Files
<FilesMatch "^(wp-config\.php|readme\.html|license\.txt|\.env|\.git|\.htaccess|\.htpasswd|debug\.log)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>

# 4. Block XML-RPC completely (Stops Brute Force Attacks & Pingback DDoS)
<Files xmlrpc.php>
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</Files>

# 5. Prevent PHP Execution in Uploads Directory
# (Place this rule inside wp-content/uploads/.htaccess as well)
<Directory "*wp-content/uploads/">
    <Files "*.php">
        <IfModule mod_authz_core.c>
            Require all denied
        </IfModule>
        <IfModule !mod_authz_core.c>
            Order allow,deny
            Deny from all
        </IfModule>
    </Files>
</Directory>

# 6. HTTP Security Headers
<IfModule mod_headers.c>
    # Prevent Clickjacking
    Header always set X-Frame-Options "SAMEORIGIN"

    # Prevent MIME-type Sniffing
    Header always set X-Content-Type-Options "nosniff"

    # XSS Protection
    Header always set X-XSS-Protection "1; mode=block"

    # Referrer Policy
    Header always set Referrer-Policy "strict-origin-when-cross-origin"

    # Restrict Hardware Permissions
    Header always set Permissions-Policy "geolocation=(), microphone=(), camera=()"

    # Enforce HSTS (1 year)
    Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
</IfModule>

# 7. Block Malicious Query Strings, SQL Injections & Script Probes
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteCond %{QUERY_STRING} (eval\(|base64_decode|GLOBALS|_REQUEST|CONCAT|UNION|SELECT.*FROM) [NC,OR]
RewriteCond %{QUERY_STRING} (<|%3C).*script.*(>|%3E) [NC,OR]
RewriteCond %{QUERY_STRING} (javascript:.*|document\.cookie) [NC]
RewriteRule .* - [F,L]
</IfModule>

# 8. High Performance Gzip Compression
<IfModule mod_deflate.c>
AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css text/javascript application/javascript application/json image/svg+xml
</IfModule>
